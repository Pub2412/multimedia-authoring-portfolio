"""
Mars + Moons 3D Model — Blender 5.x  |  Auto-Download Textures Edition
===================================================================
Requires Blender 5.0 or newer.

Models Mars, Phobos, and Deimos, each spinning on its own axis while
Phobos and Deimos orbit Mars. Background galaxy uses a CUSTOM PNG image.
Texture source: Solar System Scope (CC BY 4.0).
"""

SAVE_DIR = ""   # Leave "" to save next to the .blend file,
                # or set a full path, e.g. r"C:\Textures\Mars"

# ── Texture quality ────────────────────────────────────────────────────────
# NOTE: verify these URLs resolve before running — Wikimedia file hashes
# can differ from the Sun texture's. If a download fails, grab the
# corresponding 2k texture from solarsystemscope.com/textures manually
# and drop it in SAVE_DIR with the matching filename below.
MARS_TEXTURE_URL    = "https://upload.wikimedia.org/wikipedia/commons/8/8d/Solarsystemscope_texture_2k_mars.jpg"
MARS_TEXTURE_NAME    = "2k_mars.jpg"

PHOBOS_TEXTURE_URL   = "https://upload.wikimedia.org/wikipedia/commons/c/c7/Phobos_colour_2008.jpg"
PHOBOS_TEXTURE_NAME  = "phobos.jpg"

DEIMOS_TEXTURE_URL   = "https://upload.wikimedia.org/wikipedia/commons/4/4e/Deimos-MRO.jpg"
DEIMOS_TEXTURE_NAME  = "deimos.jpg"

# ── Scale & Animation ─────────────────────────────────────────────────────
SCALE_KM_PER_BU   = 1_000        # 1 Blender Unit = 1,000 km
MARS_RADIUS_KM    = 3_389.5
PHOBOS_RADIUS_KM  = 11.27
DEIMOS_RADIUS_KM  = 6.2
PHOBOS_ORBIT_KM   = 9_377        # mean orbital distance from Mars centre
DEIMOS_ORBIT_KM   = 23_460
PHOBOS_ORBIT_DAYS = 0.3189
DEIMOS_ORBIT_DAYS = 1.263
MARS_ROTATION_DAYS = 1.026

MOON_SIZE_EXAGGERATION = 6.0     # real moon radii are nearly invisible at this scale

ANIM_FRAMES = 360
FPS         = 24

# ── Render / Camera ────────────────────────────────────────────────────────
CAMERA_DISTANCE = 60.0
CAMERA_HEIGHT   = 18.0
CAMERA_FOCAL_MM = 35.0

# ── Custom background image (PNG) ──────────────────────────────────────────
BACKGROUND_IMAGE_PATH = r"C:\Users\Lancelot\Documents\School Files\Y2T3\MMA\Solar_System\Planets\Mars\bg.png"
BACKGROUND_STRENGTH    = 1.5
BG_IMAGE_WIDTH         = 1980
BG_IMAGE_HEIGHT        = 1080

# ─────────────────────────────────────────────────────────────────────────

import bpy
import math
import os
import urllib.request
import ssl
from bpy_extras import anim_utils

MARS_RADIUS_BU   = MARS_RADIUS_KM / SCALE_KM_PER_BU
PHOBOS_RADIUS_BU = (PHOBOS_RADIUS_KM / SCALE_KM_PER_BU) * MOON_SIZE_EXAGGERATION
DEIMOS_RADIUS_BU = (DEIMOS_RADIUS_KM / SCALE_KM_PER_BU) * MOON_SIZE_EXAGGERATION
PHOBOS_ORBIT_BU  = PHOBOS_ORBIT_KM / SCALE_KM_PER_BU
DEIMOS_ORBIT_BU  = DEIMOS_ORBIT_KM / SCALE_KM_PER_BU


# ── 0. Resolve save directory ───────────────────────────────────────────────

def get_save_dir():
    if SAVE_DIR:
        d = os.path.normpath(SAVE_DIR)
    else:
        blend_path = bpy.data.filepath
        if blend_path:
            d = os.path.dirname(blend_path)
        else:
            d = os.path.join(os.path.expanduser("~"), "Mars_Textures")
    os.makedirs(d, exist_ok=True)
    return d


# ── 1. Download texture ─────────────────────────────────────────────────────

def download_texture(url, name, save_dir):
    dest = os.path.join(save_dir, name)

    if os.path.isfile(dest):
        print(f"  [SKIP] Already exists: {name}")
        return dest

    print(f"  [DL]  Downloading {name} …")
    ctx = ssl.create_default_context()
    ctx.check_hostname = False
    ctx.verify_mode    = ssl.CERT_NONE

    try:
        req = urllib.request.Request(
            url, headers={"User-Agent": "Mozilla/5.0 (Blender texture downloader)"},
        )
        with urllib.request.urlopen(req, context=ctx) as resp, open(dest, "wb") as f:
            total = int(resp.headers.get("Content-Length", 0))
            downloaded = 0
            chunk = 1 << 16
            while True:
                data = resp.read(chunk)
                if not data:
                    break
                f.write(data)
                downloaded += len(data)
                if total:
                    pct = downloaded / total * 100
                    print(f"       {pct:5.1f}%  ({downloaded//1024} KB / {total//1024} KB)",
                          end="\r")
        print(f"\n  [OK]  Saved → {dest}")
        return dest
    except Exception as e:
        print(f"  [ERR] Failed to download {url}\n        {e}")
        return None


# ── 2. Load image ───────────────────────────────────────────────────────────

def load_image(filepath, colorspace="sRGB"):
    if not filepath or not os.path.isfile(filepath):
        return None
    img = bpy.data.images.load(filepath, check_existing=True)
    img.colorspace_settings.name = colorspace
    return img


# ── 3. Clear scene ──────────────────────────────────────────────────────────

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete(use_global=False)
    for col in bpy.data.collections:
        bpy.data.collections.remove(col)
    bpy.ops.outliner.orphans_purge(do_recursive=True)


# ── 4. Sphere ────────────────────────────────────────────────────────────────

def create_sphere(name, radius, location=(0, 0, 0), segments=128, ring_count=64):
    bpy.ops.mesh.primitive_uv_sphere_add(
        radius=radius, segments=segments, ring_count=ring_count, location=location
    )
    obj = bpy.context.active_object
    obj.name = name
    bpy.ops.object.shade_smooth()
    sub = obj.modifiers.new("Subdivision", 'SUBSURF')
    sub.levels = 2; sub.render_levels = 3
    return obj


# ── 5. Materials (non-emissive, lit surfaces) ────────────────────────────────

def create_surface_material(name, img_color, roughness=0.9):
    mat   = bpy.data.materials.new(name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    links = mat.node_tree.links
    nodes.clear()

    def N(t, loc):
        n = nodes.new(t); n.location = loc; return n

    out  = N("ShaderNodeOutputMaterial", (1000, 0))
    bsdf = N("ShaderNodeBsdfPrincipled", (700, 0))
    bsdf.inputs["Roughness"].default_value = roughness

    tex = N("ShaderNodeTexImage", (300, 0))
    if img_color:
        tex.image = img_color
        links.new(tex.outputs["Color"], bsdf.inputs["Base Color"])

    links.new(bsdf.outputs["BSDF"], out.inputs["Surface"])
    return mat


# ── 6. Axial rotation ─────────────────────────────────────────────────────────

def animate_spin(obj, period_days, frame_start=1, frame_end=ANIM_FRAMES):
    """Spins obj on its local Z axis. period_days is the real rotation period;
    the animation always completes one full spin over ANIM_FRAMES purely for
    a consistent visual loop (not literal real-time)."""
    obj.rotation_euler = (0, 0, 0.0)
    obj.keyframe_insert(data_path="rotation_euler", frame=frame_start)
    obj.rotation_euler[2] = math.tau
    obj.keyframe_insert(data_path="rotation_euler", frame=frame_end)
    _linearize(obj)


def animate_orbit(obj, radius, frame_start=1, frame_end=ANIM_FRAMES, loops=1):
    """Animates obj on a circular orbit of given radius around the world origin."""
    steps = 4  # quarter-circle keyframes for a clean circular path
    for i in range(steps + 1):
        frame = frame_start + (frame_end - frame_start) * (i / steps)
        angle = math.tau * loops * (i / steps)
        obj.location = (radius * math.cos(angle), radius * math.sin(angle), obj.location.z)
        obj.keyframe_insert(data_path="location", frame=frame)
    _linearize(obj)


def _linearize(obj):
    ad = obj.animation_data
    if ad and ad.action:
        cb = anim_utils.action_get_channelbag_for_slot(ad.action, ad.action_slot)
        if cb:
            for fc in cb.fcurves:
                for kp in fc.keyframe_points:
                    kp.interpolation = 'LINEAR'


# ── 7. Camera ──────────────────────────────────────────────────────────────────

def create_camera():
    bpy.ops.object.camera_add(location=(0, -CAMERA_DISTANCE, CAMERA_HEIGHT))
    cam = bpy.context.active_object
    cam.name = "Camera"
    cam.rotation_euler = (math.radians(73), 0, 0)
    cam.data.lens = CAMERA_FOCAL_MM
    bpy.context.scene.camera = cam


# ── 8. Sun lamp (Mars is not self-illuminating) ─────────────────────────────

def create_sun_lamp():
    bpy.ops.object.light_add(type='SUN', location=(-30, -30, 30))
    lamp = bpy.context.active_object
    lamp.name = "Sunlight"
    lamp.data.energy = 3.0
    lamp.rotation_euler = (math.radians(55), 0, math.radians(35))
    return lamp


# ── 9. World / Custom PNG background with aspect fix ────────────────────────

def setup_world():
    world = bpy.context.scene.world
    if world is None:
        world = bpy.data.worlds.new("World")
        bpy.context.scene.world = world
    world.use_nodes = True

    wn = world.node_tree.nodes
    wl = world.node_tree.links
    wn.clear()

    out = wn.new("ShaderNodeOutputWorld")
    bg  = wn.new("ShaderNodeBackground")
    bg.inputs["Strength"].default_value = BACKGROUND_STRENGTH

    tex = wn.new("ShaderNodeTexImage")
    if os.path.isfile(BACKGROUND_IMAGE_PATH):
        tex.image = bpy.data.images.load(BACKGROUND_IMAGE_PATH, check_existing=True)
        tex.image.colorspace_settings.name = "sRGB"
        print(f"  BG PNG  : Custom background → {BACKGROUND_IMAGE_PATH}")
    else:
        print("  BG PNG  : Background image not found, using black")
        bg.inputs["Color"].default_value = (0.0, 0.0, 0.0, 1)

    tc = wn.new("ShaderNodeTexCoord")
    mapping = wn.new("ShaderNodeMapping")

    aspect = BG_IMAGE_WIDTH / BG_IMAGE_HEIGHT
    mapping.inputs["Scale"].default_value = (aspect, 1.0, 1.0)

    wl.new(tc.outputs["Window"], mapping.inputs["Vector"])
    wl.new(mapping.outputs["Vector"], tex.inputs["Vector"])
    wl.new(tex.outputs["Color"], bg.inputs["Color"])
    wl.new(bg.outputs["Background"], out.inputs["Surface"])


# ── 10. Render settings ────────────────────────────────────────────────────────

def setup_render():
    scene = bpy.context.scene
    scene.render.engine          = 'CYCLES'
    scene.cycles.samples         = 128
    scene.cycles.use_denoising  = True
    scene.render.resolution_x    = 1920
    scene.render.resolution_y    = 1080
    scene.render.resolution_percentage = 100


# ── Main ─────────────────────────────────────────────────────────────────────

def main():
    save_dir = get_save_dir()

    mars_tex   = download_texture(MARS_TEXTURE_URL, MARS_TEXTURE_NAME, save_dir)
    phobos_tex = download_texture(PHOBOS_TEXTURE_URL, PHOBOS_TEXTURE_NAME, save_dir)
    deimos_tex = download_texture(DEIMOS_TEXTURE_URL, DEIMOS_TEXTURE_NAME, save_dir)

    clear_scene()

    scene = bpy.context.scene
    scene.frame_start = 1; scene.frame_end = ANIM_FRAMES; scene.render.fps = FPS

    # Mars
    mars_img = load_image(mars_tex, "sRGB")
    mars = create_sphere("Mars", MARS_RADIUS_BU)
    mars.data.materials.append(create_surface_material("Mars_Surface", mars_img, roughness=0.95))
    animate_spin(mars, MARS_ROTATION_DAYS)

    # Phobos — orbits faster (closer + shorter period) than Deimos
    phobos_img = load_image(phobos_tex, "sRGB")
    phobos = create_sphere("Phobos", PHOBOS_RADIUS_BU,
                            location=(PHOBOS_ORBIT_BU, 0, 0), segments=48, ring_count=24)
    phobos.data.materials.append(create_surface_material("Phobos_Surface", phobos_img, roughness=0.95))
    animate_spin(phobos, PHOBOS_ORBIT_DAYS)  # tidally locked: spin period == orbit period
    # Phobos orbits ~4x faster than Deimos (real ratio ≈ 0.3189/1.263 days), so it
    # completes proportionally more loops within the same ANIM_FRAMES window.
    phobos_loops = round(DEIMOS_ORBIT_DAYS / PHOBOS_ORBIT_DAYS)
    animate_orbit(phobos, PHOBOS_ORBIT_BU, loops=phobos_loops)

    # Deimos
    deimos_img = load_image(deimos_tex, "sRGB")
    deimos = create_sphere("Deimos", DEIMOS_RADIUS_BU,
                            location=(DEIMOS_ORBIT_BU, 0, 0), segments=48, ring_count=24)
    deimos.data.materials.append(create_surface_material("Deimos_Surface", deimos_img, roughness=0.95))
    animate_spin(deimos, DEIMOS_ORBIT_DAYS)
    animate_orbit(deimos, DEIMOS_ORBIT_BU, loops=1)

    create_camera()
    create_sun_lamp()
    setup_world()
    setup_render()


main()
